% Wrapper graceful : si l'image n'existe pas, afficher un placeholder cadré
% plutôt que de faire échouer la compilation. Indispensable tant que les
% schémas du DSR n'ont pas été ré-exportés dans images/.
\let\StigOriginalIncludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \IfFileExists{#2}{\StigOriginalIncludegraphics[#1]{#2}}{%
    \IfFileExists{images/#2}{\StigOriginalIncludegraphics[#1]{#2}}{%
      \fbox{\parbox{0.7\textwidth}{\centering\vspace{0.6cm}%
        \textit{[Figure \`a fournir : \texttt{\detokenize{#2}}]}%
        \vspace{0.6cm}}}%
    }%
  }%
}
